# --- Stage 1.1: Extract & Chunk (stage1_1_eda_extract_and_chunk.ipynb) ---
gdown>=6.1     # lists the public GOOGLE_DRIVE_SOURCES_FOLDER (needs skip_download=)
               # and downloads the source PDFs from Google Drive
pymupdf        # NOTE: package name is "pymupdf" but it's imported as `import fitz`
tiktoken       # token-accurate child-chunk sizing (cl100k_base encoding)
pytesseract    # source10 only: that PDF is a scanned book with NO text layer at
               # all, so it's OCR'd page-by-page. pytesseract is just a wrapper --
               # you ALSO need the Tesseract OCR engine itself installed system-side:
               # `brew install tesseract` (macOS) / `apt install tesseract-ocr` (Linux).
               # Every other source has a real text layer and never touches this.
Pillow         # source10 only: wraps the OCR page images for pytesseract
pdfplumber     # source15/source16 only: extracts real data tables (nutrient
               # tables, etc.) as their own chunk_type="table" child chunks
               # instead of flattening them into prose -- pdfplumber's table-
               # grid detection is far more reliable here than reconstructing
               # rows/columns from fitz's plain text. common.py's
               # extract_page_range_tables() wraps every page in a wall-clock
               # timeout since pdfminer (which pdfplumber sits on) can
               # pathologically hang on a small minority of real pages.
